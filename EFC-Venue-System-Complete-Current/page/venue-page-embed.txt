<script>

window.addEventListener(
    "load",
    function () {

        const container =
            document.getElementById(
                "efc-venue-history"
            );

        if (
            typeof EFC === "undefined" ||
            typeof EFC.renderVenueHistoryPage !==
            "function"
        ) {
            container.innerHTML =
                "<p class='efc-venue-page-loading'>Venue history library not loaded.</p>";

            return;
        }

        EFC.renderVenueHistoryPage(
            "efc-venue-history",
            {
                title:
                    "Where We've Played",
                description:
                    "Explore the venues, events, and private engagements that have shaped Electric Flower Co.'s performance history."
            }
        );

    }
);

</script>
